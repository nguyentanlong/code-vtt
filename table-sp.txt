CREATE   PROCEDURE dbo.sp_v2_NhanVien_GetList
    @Keyword NVARCHAR(255) = NULL,
    @ChiNhanhID INT = NULL,
    @PhongBanID INT = NULL,
    @TrangThai TINYINT = NULL,
    @PageNumber INT = 1,
    @PageSize INT = 12
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TongSoDong
    FROM dbo.NhanVien nv
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = nv.PhongBanChinhThucID
    WHERE nv.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR nv.MaNhanVien LIKE '%' + @Keyword + '%' OR nv.HoTen LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR nv.PhongBanChinhThucID = @PhongBanID)
      AND (@TrangThai IS NULL OR nv.TrangThai = @TrangThai);

    SELECT 
        nv.NhanVienID, nv.HoTen, nv.MaNhanVien, nv.Email, nv.SoDienThoai,
        nv.PhongBanChinhThucID, pb.TenPhongBan, pb.ChiNhanhID, cn.TenChiNhanh,
        nv.ChucVuChinhThucID, cv.TenChucVu, cv.CapBac,
        nv.TrangThai
    FROM dbo.NhanVien nv
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = nv.PhongBanChinhThucID
    LEFT JOIN dbo.ChiNhanh cn ON cn.ChiNhanhID = pb.ChiNhanhID
    LEFT JOIN dbo.ChucVu cv ON cv.ChucVuID = nv.ChucVuChinhThucID
    WHERE nv.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR nv.MaNhanVien LIKE '%' + @Keyword + '%' OR nv.HoTen LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR nv.PhongBanChinhThucID = @PhongBanID)
      AND (@TrangThai IS NULL OR nv.TrangThai = @TrangThai)
    ORDER BY nv.NhanVienID DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

GO

-- 1.3. Xóa mềm Phòng ban (Soft Delete)
CREATE   PROCEDURE dbo.sp_chinh_DMPhongBan_Delete
    @PhongBanID BIGINT,
    @ResponseCode INT OUTPUT,
    @ResponseMessage NVARCHAR(255) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Kiểm tra xem có phòng ban con không
    IF EXISTS (SELECT 1 FROM dbo.DMPhongBan WHERE PhongBanChaID = @PhongBanID AND IsDeleted = 0)
    BEGIN
        SET @ResponseCode = -1;
        SET @ResponseMessage = N'Không thể xóa vì phòng ban này đang chứa các phòng ban con!';
        RETURN;
    END

    -- Kiểm tra xem có bộ phận thuộc phòng ban này không
    IF EXISTS (SELECT 1 FROM dbo.DMBoPhan WHERE PhongBanID = @PhongBanID AND IsDeleted = 0)
    BEGIN
        SET @ResponseCode = -2;
        SET @ResponseMessage = N'Không thể xóa vì phòng ban này đang có các Bộ phận trực thuộc!';
        RETURN;
    END

    UPDATE dbo.DMPhongBan 
    SET IsDeleted = 1 
    WHERE PhongBanID = @PhongBanID;

    SET @ResponseCode = 1;
    SET @ResponseMessage = N'Xóa phòng ban thành công!';
END;

GO

CREATE PROCEDURE [dbo].[sp_chinh_DMNhanVien_GetList]
    @CongTyID BIGINT,
    @Keyword NVARCHAR(100) = NULL,
    @TrangThai TINYINT = 1 -- 1: Đang làm việc, 0: Ngừng hoạt động, NULL: Tất cả
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        nv.NhanVienID,
        nv.MaNhanVien,
        nv.HoTen,
        nv.PhongBanID,
        ISNULL(pb.TenPhongBan, N'Chưa xếp phòng') AS TenPhongBan,
        bp.BoPhanID,
        ISNULL(bp.TenBoPhan, N'') AS TenBoPhan,
        nv.TrangThai
    FROM 
        dbo.DMNhanVien nv WITH (NOLOCK)
    LEFT JOIN 
        dbo.DMPhongBan pb WITH (NOLOCK) 
        ON nv.PhongBanID = pb.PhongBanID AND pb.IsDeleted = 0
    LEFT JOIN 
        dbo.DMBoPhan bp WITH (NOLOCK) 
        ON bp.PhongBanID = nv.PhongBanID AND bp.TruongBoPhanID = nv.NhanVienID AND bp.IsDeleted = 0
    WHERE 
        nv.CongTyID = @CongTyID
        AND nv.IsDeleted = 0
        AND (@TrangThai IS NULL OR nv.TrangThai = @TrangThai)
        AND (
            @Keyword IS NULL 
            OR @Keyword = '' 
            OR nv.MaNhanVien LIKE '%' + @Keyword + '%'
            OR nv.HoTen LIKE N'%' + @Keyword + '%'
            OR pb.TenPhongBan LIKE N'%' + @Keyword + '%'
            OR bp.TenBoPhan LIKE N'%' + @Keyword + '%'
        )
    ORDER BY 
        pb.TenPhongBan ASC,  -- Gom nhóm theo Phòng Ban
        bp.TenBoPhan ASC,   -- Gom nhóm theo Bộ Phận
        nv.HoTen ASC;       -- Sắp xếp theo Tên Nhân Viên
END

GO

BEGIN

-- 1.2. Thêm mới/Cập nhật Phòng ban (Upsert)
CREATE   PROCEDURE dbo.sp_chinh_DMPhongBan_Save
    @PhongBanID BIGINT = 0, -- 0: Insert, >0: Update
    @CongTyID BIGINT,
    @MaPhongBan VARCHAR(30),
    @TenPhongBan NVARCHAR(255),
    @PhongBanChaID BIGINT = NULL,
    @TruongPhongID BIGINT = NULL,
    @ThuTu INT = 0,
    @TrangThai TINYINT = 1,
    @ResponseCode INT OUTPUT, -- 1: Success, -1: Duplicate Code, -2: Parent Not Found
    @ResponseMessage NVARCHAR(255) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        -- Check trùng MaPhongBan trong cùng CongTyID
        IF EXISTS (
            SELECT 1 FROM dbo.DMPhongBan 
            WHERE CongTyID = @CongTyID 
              AND MaPhongBan = @MaPhongBan 
              AND PhongBanID <> @PhongBanID 
              AND IsDeleted = 0
        )
        BEGIN
            SET @ResponseCode = -1;
            SET @ResponseMessage = N'Mã phòng ban đã tồn tại trong công ty này!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Tính CapDo và DuongDan dựa trên PhongBanChaID
        DECLARE @CapDo INT = 1;
        DECLARE @DuongDan NVARCHAR(500) = '';

        IF @PhongBanChaID IS NOT NULL AND @PhongBanChaID > 0
        BEGIN
            SELECT @CapDo = CapDo + 1, @DuongDan = ISNULL(DuongDan, '')
            FROM dbo.DMPhongBan 
            WHERE PhongBanID = @PhongBanChaID AND IsDeleted = 0;

            IF @@ROWCOUNT = 0
            BEGIN
                SET @ResponseCode = -2;
                SET @ResponseMessage = N'Phòng ban cấp cha không tồn tại!';
                ROLLBACK TRANSACTION;
                RETURN;
            END
        END

        IF @PhongBanID = 0
        BEGIN
            -- INSERT
            INSERT INTO dbo.DMPhongBan (
                CongTyID, MaPhongBan, TenPhongBan, PhongBanChaID, 
                CapDo, DuongDan, TruongPhongID, ThuTu, TrangThai, NgayTao, IsDeleted, Version
            )
            VALUES (
                @CongTyID, @MaPhongBan, @TenPhongBan, @PhongBanChaID, 
                @CapDo, '', @TruongPhongID, @ThuTu, @TrangThai, SYSUTCDATETIME(), 0, 1
            );

            SET @PhongBanID = SCOPE_IDENTITY();
            
            -- Update lại Đường dẫn dạng: /1/4/12/
            SET @DuongDan = CASE 
                WHEN @DuongDan = '' THEN '/' + CAST(@PhongBanID AS VARCHAR) + '/'
                ELSE @DuongDan + CAST(@PhongBanID AS VARCHAR) + '/' 
            END;

            UPDATE dbo.DMPhongBan 
            SET DuongDan = @DuongDan 
            WHERE PhongBanID = @PhongBanID;
        END
        ELSE
        BEGIN
            -- UPDATE
            SET @DuongDan = CASE 
                WHEN @DuongDan = '' THEN '/' + CAST(@PhongBanID AS VARCHAR) + '/'
                ELSE @DuongDan + CAST(@PhongBanID AS VARCHAR) + '/' 
            END;

            UPDATE dbo.DMPhongBan
            SET MaPhongBan = @MaPhongBan,
                TenPhongBan = @TenPhongBan,
                PhongBanChaID = @PhongBanChaID,
                CapDo = @CapDo,
                DuongDan = @DuongDan,
                TruongPhongID = @TruongPhongID,
                ThuTu = @ThuTu,
                TrangThai = @TrangThai,
                Version = Version + 1
            WHERE PhongBanID = @PhongBanID AND CongTyID = @CongTyID AND IsDeleted = 0;
        END

        COMMIT TRANSACTION;
        SET @ResponseCode = 1;
        SET @ResponseMessage = N'Thao tác thành công!';
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SET @ResponseCode = 99;
        SET @ResponseMessage = ERROR_MESSAGE();
    END CATCH
END;

GO

CREATE   PROCEDURE dbo.sp_v2_TaiKhoan_GetList
    @Keyword NVARCHAR(255) = NULL,
    @ChiNhanhID INT = NULL,
    @PhongBanID INT = NULL,
    @PageNumber INT = 1,
    @PageSize INT = 12
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TongSoDong
    FROM dbo.TaiKhoan tk
    INNER JOIN dbo.NhanVien nv ON nv.NhanVienID = tk.NhanVienID
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = nv.PhongBanChinhThucID
    WHERE (@Keyword IS NULL OR @Keyword = '' OR tk.TenDangNhap LIKE '%' + @Keyword + '%' OR nv.HoTen LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR nv.PhongBanChinhThucID = @PhongBanID);

    SELECT 
        tk.TaiKhoanID, tk.TenDangNhap, tk.TrangThai, tk.IsLocked,
        nv.NhanVienID, nv.HoTen, nv.PhongBanChinhThucID, pb.TenPhongBan, pb.ChiNhanhID, cn.TenChiNhanh,
        (
            SELECT TOP 1 nq.TenNhomQuyen
            FROM dbo.GanNhomQuyen g
            INNER JOIN dbo.NhomQuyen nq ON nq.NhomQuyenID = g.NhomQuyenID
            LEFT JOIN dbo.PhanCongNhanVien pcn ON pcn.PhanCongID = g.PhanCongID
            WHERE g.TrangThai = 1
              AND (
                    (g.ScopeType = 'TAIKHOAN' AND g.TaiKhoanID = tk.TaiKhoanID)
                 OR (g.ScopeType = 'PHANCONG' AND pcn.NhanVienID = nv.NhanVienID AND pcn.TrangThai = 1)
              )
            ORDER BY nq.CapBacTuongUng ASC
        ) AS TenVaiTro
    FROM dbo.TaiKhoan tk
    INNER JOIN dbo.NhanVien nv ON nv.NhanVienID = tk.NhanVienID
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = nv.PhongBanChinhThucID
    LEFT JOIN dbo.ChiNhanh cn ON cn.ChiNhanhID = pb.ChiNhanhID
    WHERE (@Keyword IS NULL OR @Keyword = '' OR tk.TenDangNhap LIKE '%' + @Keyword + '%' OR nv.HoTen LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR nv.PhongBanChinhThucID = @PhongBanID)
    ORDER BY tk.TaiKhoanID DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

CREATE   PROCEDURE dbo.sp_chinh_DMBoPhan_Save
    @BoPhanID BIGINT = 0,
    @PhongBanID BIGINT,
    @MaBoPhan VARCHAR(30),
    @TenBoPhan NVARCHAR(255),
    @TruongBoPhanID BIGINT = NULL,
    @ThuTu INT = 0,
    @TrangThai TINYINT = 1
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @ResponseCode INT = 0;
    DECLARE @ResponseMessage NVARCHAR(255) = N'';

    BEGIN TRY
        -- 1. Kiểm tra trùng mã bộ phận trong cùng phòng ban
        IF EXISTS (
            SELECT 1 
            FROM dbo.DMBoPhan 
            WHERE MaBoPhan = @MaBoPhan 
              AND PhongBanID = @PhongBanID 
              AND BoPhanID <> @BoPhanID 
              AND IsDeleted = 0
        )
        BEGIN
            SET @ResponseCode = -1;
            SET @ResponseMessage = N'Mã bộ phận đã tồn tại trong phòng ban này!';
        END
        ELSE IF @BoPhanID = 0
        BEGIN
            -- 2. Thêm mới
            INSERT INTO dbo.DMBoPhan (
                PhongBanID, MaBoPhan, TenBoPhan, TruongBoPhanID, ThuTu, TrangThai, NgayTao, IsDeleted
            )
            VALUES (
                @PhongBanID, @MaBoPhan, @TenBoPhan, @TruongBoPhanID, @ThuTu, @TrangThai, SYSUTCDATETIME(), 0
            );

            SET @ResponseCode = 1;
            SET @ResponseMessage = N'Thêm mới bộ phận thành công!';
        END
        ELSE
        BEGIN
            -- 3. Cập nhật
            UPDATE dbo.DMBoPhan
            SET PhongBanID = @PhongBanID,
                MaBoPhan = @MaBoPhan,
                TenBoPhan = @TenBoPhan,
                TruongBoPhanID = @TruongBoPhanID,
                ThuTu = @ThuTu,
                TrangThai = @TrangThai
            WHERE BoPhanID = @BoPhanID AND IsDeleted = 0;

            SET @ResponseCode = 1;
            SET @ResponseMessage = N'Cập nhật bộ phận thành công!';
        END
    END TRY
    BEGIN CATCH
        SET @ResponseCode = 0;
        SET @ResponseMessage = ERROR_MESSAGE();
    END CATCH;

    -- TRẢ VỀ DATASET CHO ConnectServer LẤY DỮ LIỆU (KHÔNG DÙNG OUTPUT PARAMETER)
    SELECT @ResponseCode AS ResponseCode, @ResponseMessage AS ResponseMessage;
END;

GO

CREATE PROCEDURE dbo.sp_chinh_TaiKhoan_OTP
    @Action VARCHAR(20),             -- 'GENERATE', 'VERIFY'
    @TaiKhoanID BIGINT,
    @OTPCode VARCHAR(10) = NULL,
    @IPAddress VARCHAR(50) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. ACTION: GENERATE (Chỉ giữ lại 1 OTP active duy nhất)
    IF (@Action = 'GENERATE')
    BEGIN
        -- HỦY BỎ TẤT CẢ OTP CŨ CỦA TÀI KHOẢN NÀY (Chính sách 1 Email - 1 OTP)
        UPDATE dbo.NhatKyHeThong 
        SET LoaiLog = 'OTP_EXPIRED' 
        WHERE NguoiThucHienID = @TaiKhoanID AND LoaiLog = 'OTP_PENDING';

        -- Tạo OTP mới
        INSERT INTO dbo.NhatKyHeThong (LoaiLog, NoiDung, NguoiThucHienID, IP, ThoiGian)
        VALUES ('OTP_PENDING', @OTPCode, @TaiKhoanID, @IPAddress, SYSUTCDATETIME());

        SELECT 1 AS Success, N'Đã khởi tạo mã OTP mới' AS Message;
        RETURN;
    END

    -- 2. ACTION: VERIFY (Kiểm tra mã OTP mới nhất)
    IF (@Action = 'VERIFY')
    BEGIN
        DECLARE @ValidOTP VARCHAR(10);
        DECLARE @CreatedTime DATETIME2;

        -- Chỉ lấy mã PENDING duy nhất/mới nhất
        SELECT TOP 1 
            @ValidOTP = NoiDung,
            @CreatedTime = ThoiGian
        FROM dbo.NhatKyHeThong WITH(NOLOCK)
        WHERE NguoiThucHienID = @TaiKhoanID 
          AND LoaiLog = 'OTP_PENDING'
        ORDER BY LogID DESC;

        -- Nếu không có OTP nào đang active hoặc OTP nhập không khớp
        IF (@ValidOTP IS NULL OR @ValidOTP <> @OTPCode)
        BEGIN
            SELECT 0 AS Success, N'Mã OTP không chính xác hoặc đã bị vô hiệu hóa!' AS Message;
            RETURN;
        END

        -- Kiểm tra thời hạn 5 phút (300 giây)
        IF (DATEDIFF(SECOND, @CreatedTime, SYSUTCDATETIME()) > 300)
        BEGIN
            UPDATE dbo.NhatKyHeThong SET LoaiLog = 'OTP_EXPIRED' WHERE NguoiThucHienID = @TaiKhoanID AND LoaiLog = 'OTP_PENDING';
            SELECT 0 AS Success, N'Mã OTP đã hết hạn! Vui lòng lấy mã mới.' AS Message;
            RETURN;
        END

        -- OTP Hợp lệ -> Đánh dấu đã dùng thành công
        UPDATE dbo.NhatKyHeThong 
        SET LoaiLog = 'OTP_VERIFIED' 
        WHERE NguoiThucHienID = @TaiKhoanID AND LoaiLog = 'OTP_PENDING';

        SELECT 1 AS Success, N'Xác thực OTP thành công!' AS Message;
        RETURN;
    END
END

GO


CREATE   PROCEDURE dbo.sp_long_DMLoaiCongTrinh_Delete
    @LoaiCongTrinhID BIGINT
AS

CREATE   PROCEDURE dbo.sp_v2_PhongBan_GetList
    @Keyword NVARCHAR(255) = NULL,
    @ChiNhanhID INT = NULL,
    @PhongBanID INT = NULL,
    @TrangThai TINYINT = NULL,

    @PageNumber INT = 1,
    @PageSize INT = 12
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TongSoDong
    FROM dbo.PhongBan pb
    WHERE pb.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR pb.MaPhongBan LIKE '%' + @Keyword + '%' OR pb.TenPhongBan LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR pb.PhongBanID = @PhongBanID OR pb.PhongBanChaID = @PhongBanID)
      AND (@TrangThai IS NULL OR pb.TrangThai = @TrangThai);

    SELECT 
        pb.PhongBanID, pb.ChiNhanhID, cn.TenChiNhanh,
        pb.PhongBanChaID, cha.TenPhongBan AS TenPhongBanCha,
        pb.MaPhongBan, pb.TenPhongBan, pb.CapDo, pb.ThuTu, pb.TrangThai
    FROM dbo.PhongBan pb
    LEFT JOIN dbo.ChiNhanh cn ON cn.ChiNhanhID = pb.ChiNhanhID
    LEFT JOIN dbo.PhongBan cha ON cha.PhongBanID = pb.PhongBanChaID
    WHERE pb.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR pb.MaPhongBan LIKE '%' + @Keyword + '%' OR pb.TenPhongBan LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR pb.PhongBanID = @PhongBanID OR pb.PhongBanChaID = @PhongBanID)
      AND (@TrangThai IS NULL OR pb.TrangThai = @TrangThai)
    ORDER BY pb.CapDo ASC, pb.ThuTu ASC, pb.TenPhongBan ASC
    OFFSET (@PageNumber - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

GO

CREATE PROCEDURE dbo.sp_chinh_NhatKy_DangNhap
    @LoaiLog VARCHAR(50),          -- 'LOGIN_SUCCESS' hoặc 'LOGIN_FAILED'
    @Username VARCHAR(100),
    @TaiKhoanID BIGINT = NULL,
    @NhanVienID BIGINT = NULL,
    @IPAddress VARCHAR(50),
    @GhiChu NVARCHAR(255) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Nếu Đăng nhập THÀNH CÔNG -> Cập nhật LastLogin và Ghi Log Success
    IF (@LoaiLog = 'LOGIN_SUCCESS')
    BEGIN
        IF (@TaiKhoanID IS NOT NULL)
        BEGIN
            UPDATE dbo.DMTaiKhoan 
            SET LastLogin = SYSUTCDATETIME() 
            WHERE TaiKhoanID = @TaiKhoanID;
        END

        INSERT INTO dbo.NhatKyHeThong (LoaiLog, NoiDung, NguoiThucHienID, IP, ThoiGian)
        VALUES ('LOGIN_SUCCESS', N'Đăng nhập thành công', @NhanVienID, @IPAddress, SYSUTCDATETIME());
    END
    
    -- 2. Nếu Đăng nhập THẤT BẠI -> Ghi Log Failed
    ELSE IF (@LoaiLog = 'LOGIN_FAILED')
    BEGIN
        DECLARE @NoiDung NVARCHAR(500) = N'Đăng nhập thất bại: ' + ISNULL(@GhiChu, N'Sai thông tin') + N' [' + ISNULL(@Username, '') + N']';

        INSERT INTO dbo.NhatKyHeThong (LoaiLog, NoiDung, NguoiThucHienID, IP, ThoiGian)
        VALUES ('LOGIN_FAILED', @NoiDung, @NhanVienID, @IPAddress, SYSUTCDATETIME());
    END
END

GO

CREATE PROCEDURE dbo.sp_chinh_TaiKhoan_DangNhap
    @Username VARCHAR(100),
    @IPAddress VARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MaxFailedAttempts INT = 5;
    DECLARE @BlockMinutes INT = 5;
    DECLARE @FailedCount INT = 0;

    SELECT @FailedCount = ISNULL(COUNT(*), 0)
    FROM dbo.NhatKyHeThong WITH(NOLOCK)
    WHERE IP = @IPAddress AND LoaiLog = 'LOGIN_FAILED'
      AND ThoiGian >= DATEADD(MINUTE, -@BlockMinutes, SYSUTCDATETIME());

    IF (@FailedCount >= @MaxFailedAttempts)
    BEGIN
        SELECT 0 AS Success, 
               N'IP của bạn đã bị khóa ' + CAST(@BlockMinutes AS NVARCHAR(10)) + N' phút do đăng nhập sai quá 5 lần liên tiếp!' AS Message;
        RETURN;
    END

    SELECT TOP 1 
        1 AS Success,
        N'Lấy thông tin tài khoản thành công' AS Message,
        tk.TaiKhoanID, tk.Username, tk.PasswordHash, tk.IsLocked, tk.TrangThai AS TrangThaiTK,
        nv.NhanVienID, nv.HoTen, nv.EmailCongTy, nv.CongTyID, nv.PhongBanID, nv.ChucDanhID,
        vt.VaiTroID, vt.MaVaiTro, vt.TenVaiTro
    FROM dbo.DMTaiKhoan tk WITH(NOLOCK)
    LEFT JOIN dbo.DMNhanVien nv WITH(NOLOCK) ON tk.NhanVienID = nv.NhanVienID
    LEFT JOIN dbo.TaiKhoan_VaiTro tkv WITH(NOLOCK) ON tkv.TaiKhoanID = tk.TaiKhoanID AND tkv.DangHoatDong = 1
    LEFT JOIN dbo.DMVaiTro vt WITH(NOLOCK) ON vt.VaiTroID = tkv.VaiTroID
    WHERE tk.Username = @Username AND tk.IsDeleted = 0;
END

CREATE   PROCEDURE dbo.sp_v2_DuAn_GetList
    @Keyword NVARCHAR(255) = NULL,
    @ChiNhanhID INT = NULL,
    @PhongBanID INT = NULL,
    @TrangThai TINYINT = NULL,
    @PageNumber INT = 1,
    @PageSize INT = 12
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TongSoDong
    FROM dbo.DuAn da
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = da.PhongBanID
    WHERE da.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR da.MaDuAn LIKE '%' + @Keyword + '%' OR da.TenDuAn LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR da.PhongBanID = @PhongBanID)
      AND (@TrangThai IS NULL OR da.TrangThaiDuAn = @TrangThai);

    SELECT 
        da.DuAnID, da.CongTyID, da.PhongBanID, pb.TenPhongBan, pb.ChiNhanhID, cn.TenChiNhanh,
        da.LoaiCongTrinhID, lct.TenLoai,
        da.MaDuAn, da.TenDuAn, da.ChuDauTu, da.DiaDiem,
        da.NgayBatDau, da.NgayKetThucDuKien, da.TienDo, da.TrangThaiDuAn, da.NguoiTaoID
    FROM dbo.DuAn da
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = da.PhongBanID
    LEFT JOIN dbo.ChiNhanh cn ON cn.ChiNhanhID = pb.ChiNhanhID
    LEFT JOIN dbo.DMLoaiCongTrinh lct ON lct.LoaiCongTrinhID = da.LoaiCongTrinhID
    WHERE da.IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR da.MaDuAn LIKE '%' + @Keyword + '%' OR da.TenDuAn LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR da.PhongBanID = @PhongBanID)
      AND (@TrangThai IS NULL OR da.TrangThaiDuAn = @TrangThai)
    ORDER BY da.DuAnID DESC
    OFFSET (@PageNumber - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

GO

CREATE   PROCEDURE dbo.sp_v2_TaiKhoan_DangNhap
    @TenDangNhap VARCHAR(100),
    @IPAddress VARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MaxFailedShortTerm INT = 3;   -- Tầng 1: tối đa 3 lần sai
    DECLARE @ShortTermMinutes INT = 15;    -- trong vòng 15 phút -> khóa 15 phút

    DECLARE @MaxFailedDaily INT = 7;       -- Tầng 2: tối đa 7 lần sai
    DECLARE @DailyHours INT = 24;          -- trong vòng 24 giờ -> khóa tới hết ngày hôm nay

    DECLARE @FailedCountShort INT = 0;
    DECLARE @FailedCountDaily INT = 0;

    -- Tầng 1: kiểm tra 15 phút gần nhất
    SELECT @FailedCountShort = ISNULL(COUNT(*), 0)
    FROM dbo.NhatKyHeThong WITH(NOLOCK)
    WHERE IP = @IPAddress AND LoaiLog = 'LOGIN_FAILED'
      AND ThoiGian >= DATEADD(MINUTE, -@ShortTermMinutes, SYSUTCDATETIME());

    IF (@FailedCountShort >= @MaxFailedShortTerm)
    BEGIN
        SELECT 0 AS Success, 
               N'IP của bạn đã bị khóa ' + CAST(@ShortTermMinutes AS NVARCHAR(10)) + N' phút do đăng nhập sai quá ' + CAST(@MaxFailedShortTerm AS NVARCHAR(10)) + N' lần liên tiếp!' AS Message;
        RETURN;
    END

    -- Tầng 2: kiểm tra 24 giờ gần nhất
    SELECT @FailedCountDaily = ISNULL(COUNT(*), 0)
    FROM dbo.NhatKyHeThong WITH(NOLOCK)
    WHERE IP = @IPAddress AND LoaiLog = 'LOGIN_FAILED'
      AND ThoiGian >= DATEADD(HOUR, -@DailyHours, SYSUTCDATETIME());

    IF (@FailedCountDaily >= @MaxFailedDaily)
    BEGIN
        SELECT 0 AS Success, 
               N'IP của bạn đã bị khóa do đăng nhập sai quá ' + CAST(@MaxFailedDaily AS NVARCHAR(10)) + N' lần trong ngày! Vui lòng thử lại sau 24 giờ hoặc liên hệ Quản trị viên.' AS Message;
        RETURN;
    END

    -- Lấy thông tin tài khoản (không đổi so với bản trước)
    SELECT TOP 1 
        1 AS Success,
        N'Lấy thông tin tài khoản thành công' AS Message,
        tk.TaiKhoanID, tk.TenDangNhap, tk.MatKhauHash, tk.IsLocked, tk.TrangThai AS TrangThaiTK,
        nv.NhanVienID, nv.HoTen, nv.Email, nv.CongTyID, nv.PhongBanChinhThucID, nv.ChucVuChinhThucID,
        pb.ChiNhanhID
    FROM dbo.TaiKhoan tk WITH(NOLOCK)
    INNER JOIN dbo.NhanVien nv WITH(NOLOCK) ON tk.NhanVienID = nv.NhanVienID
    LEFT JOIN dbo.PhongBan pb WITH(NOLOCK) ON pb.PhongBanID = nv.PhongBanChinhThucID
    WHERE tk.TenDangNhap = @TenDangNhap AND nv.IsDeleted = 0;
END

GO

-- 3. Sửa SP kiểm tra đăng nhập — ghi đúng giờ Việt Nam khi INSERT/UPDATE
CREATE   PROCEDURE dbo.sp_v2_ThietBi_KiemTraDangNhap
    @TaiKhoanID INT,
    @DeviceID VARCHAR(100),
    @TenThietBi NVARCHAR(255),
    @IPAddress VARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MaxThietBi INT = 3;
    DECLARE @GioVN DATETIME2(0) = DATEADD(HOUR, 7, SYSUTCDATETIME());

    IF EXISTS (SELECT 1 FROM dbo.TaiKhoan_ThietBi WHERE TaiKhoanID = @TaiKhoanID AND DeviceID = @DeviceID AND DangHoatDong = 1)
    BEGIN
        UPDATE dbo.TaiKhoan_ThietBi
        SET LanHoatDongCuoi = @GioVN, IPDangNhap = @IPAddress, TenThietBi = @TenThietBi
        WHERE TaiKhoanID = @TaiKhoanID AND DeviceID = @DeviceID;

        SELECT 1 AS ChoPhep, N'' AS Message;
        RETURN;
    END

    DECLARE @SoLuongHienTai INT;
    SELECT @SoLuongHienTai = COUNT(*) FROM dbo.TaiKhoan_ThietBi WHERE TaiKhoanID = @TaiKhoanID AND DangHoatDong = 1;

    IF (@SoLuongHienTai >= @MaxThietBi)
    BEGIN
        SELECT 0 AS ChoPhep, N'Tài khoản đã đăng nhập đủ ' + CAST(@MaxThietBi AS NVARCHAR(5)) + N' thiết bị tối đa. Vui lòng đăng xuất bớt 1 thiết bị để tiếp tục!' AS Message;
        RETURN;
    END

    INSERT INTO dbo.TaiKhoan_ThietBi (TaiKhoanID, DeviceID, TenThietBi, IPDangNhap, LanDangNhapDau, LanHoatDongCuoi, DangHoatDong)
    VALUES (@TaiKhoanID, @DeviceID, @TenThietBi, @IPAddress, @GioVN, @GioVN, 1);

    SELECT 1 AS ChoPhep, N'' AS Message;
END;

GO

        RAISERROR(N'Mã tác nhân AI đã tồn tại trong công ty này!', 16, 1);
        RETURN;
    END

    IF @AIAgentID = 0
    BEGIN
        INSERT INTO dbo.AIAgent (CongTyID, PhongBanID, MaAgent, TenAgent, ModelMacDinhID, TrangThai)
        VALUES (@CongTyID, @PhongBanID, @MaAgent, @TenAgent, @ModelMacDinhID, @TrangThai);
    END
    ELSE
    BEGIN
        UPDATE dbo.AIAgent
        SET PhongBanID = @PhongBanID,
            MaAgent = @MaAgent,
            TenAgent = @TenAgent,
            ModelMacDinhID = @ModelMacDinhID,
            TrangThai = @TrangThai
        WHERE AIAgentID = @AIAgentID AND CongTyID = @CongTyID;

CREATE   PROCEDURE dbo.sp_v2_KiemTraQuyen
    @TaiKhoanID INT,
    @MaTrang VARCHAR(100),
    @MaChucNang VARCHAR(100)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @NhanVienID INT, @CongTyID INT, @PhongBanID INT, @ChiNhanhID INT;
    DECLARE @GiaTriTam VARCHAR(10) = NULL;
    DECLARE @DataScopeTam VARCHAR(30) = NULL;

    SELECT @NhanVienID = nv.NhanVienID, @CongTyID = nv.CongTyID,
           @PhongBanID = nv.PhongBanChinhThucID, @ChiNhanhID = pb.ChiNhanhID
    FROM dbo.TaiKhoan tk
    INNER JOIN dbo.NhanVien nv ON nv.NhanVienID = tk.NhanVienID
    LEFT JOIN dbo.PhongBan pb ON pb.PhongBanID = nv.PhongBanChinhThucID
    WHERE tk.TaiKhoanID = @TaiKhoanID;

    IF @NhanVienID IS NULL
    BEGIN
        SELECT CAST('DENY' AS VARCHAR(10)) AS GiaTri, CAST(NULL AS VARCHAR(30)) AS DataScope;
        RETURN;
    END

    ;WITH NguonQuyen AS (
        SELECT gnq.ScopeType, lpv.Priority, nqq.GiaTri, nqq.DataScope
        FROM dbo.GanNhomQuyen gnq
        INNER JOIN dbo.LoaiPhamVi lpv ON lpv.ScopeType = gnq.ScopeType
        INNER JOIN dbo.NhomQuyen_Quyen nqq ON nqq.NhomQuyenID = gnq.NhomQuyenID
        INNER JOIN dbo.Quyen q ON q.QuyenID = nqq.QuyenID
        INNER JOIN dbo.ChucNang cn ON cn.ChucNangID = q.ChucNangID

        INNER JOIN dbo.Trang tr ON tr.TrangID = cn.TrangID
        WHERE tr.MaTrang = @MaTrang
          AND cn.MaChucNang = @MaChucNang
          AND gnq.TrangThai = 1
          AND (gnq.TuNgay IS NULL OR gnq.TuNgay <= SYSUTCDATETIME())
          AND (gnq.DenNgay IS NULL OR gnq.DenNgay >= SYSUTCDATETIME())
          AND (
                (gnq.ScopeType = 'CONGTY' AND gnq.CongTyID = @CongTyID)
             OR (gnq.ScopeType = 'CHINHANH' AND gnq.ChiNhanhID = @ChiNhanhID)
             OR (gnq.ScopeType = 'PHONGBAN' AND gnq.PhongBanID = @PhongBanID)
             OR (gnq.ScopeType = 'PHANCONG' AND gnq.PhanCongID IN (
                    SELECT PhanCongID FROM dbo.PhanCongNhanVien
                    WHERE NhanVienID = @NhanVienID AND TrangThai = 1
                      AND (DenNgay IS NULL OR DenNgay >= CAST(GETDATE() AS DATE))
                ))
             OR (gnq.ScopeType = 'TAIKHOAN' AND gnq.TaiKhoanID = @TaiKhoanID)
          )
    )
    SELECT TOP 1 @GiaTriTam = GiaTri, @DataScopeTam = DataScope
    FROM NguonQuyen
    ORDER BY Priority DESC, CASE WHEN GiaTri = 'DENY' THEN 0 ELSE 1 END ASC;  -- SỬA: cùng Priority thì DENY thắng ALLOW

    -- TaiKhoan_Quyen (quyền riêng tài khoản) LUÔN ghi đè nếu có bản ghi hiệu lực
    SELECT TOP 1 @GiaTriTam = tkq.GiaTri, @DataScopeTam = ISNULL(tkq.DataScope, @DataScopeTam)
    FROM dbo.TaiKhoan_Quyen tkq
    INNER JOIN dbo.Quyen q ON q.QuyenID = tkq.QuyenID
    INNER JOIN dbo.ChucNang cn ON cn.ChucNangID = q.ChucNangID
    INNER JOIN dbo.Trang tr ON tr.TrangID = cn.TrangID
    WHERE tkq.TaiKhoanID = @TaiKhoanID
      AND tr.MaTrang = @MaTrang
      AND cn.MaChucNang = @MaChucNang
      AND tkq.TrangThai = 1
      AND (tkq.TuNgay IS NULL OR tkq.TuNgay <= SYSUTCDATETIME())
      AND (tkq.DenNgay IS NULL OR tkq.DenNgay >= SYSUTCDATETIME());

    IF @DataScopeTam IS NULL
    BEGIN
        SELECT TOP 1 @DataScopeTam = q.DataScope
        FROM dbo.Quyen q
        INNER JOIN dbo.ChucNang cn ON cn.ChucNangID = q.ChucNangID
        INNER JOIN dbo.Trang tr ON tr.TrangID = cn.TrangID
        WHERE tr.MaTrang = @MaTrang AND cn.MaChucNang = @MaChucNang;
    END

    SELECT 
        CAST(ISNULL(@GiaTriTam, 'DENY') AS VARCHAR(10)) AS GiaTri,
        CAST(@DataScopeTam AS VARCHAR(30)) AS DataScope;
END

GO

CREATE   PROCEDURE dbo.sp_long_DMChucDanh_Save
    @ChucDanhID BIGINT = 0,
    @CongTyID BIGINT,
    @MaChucDanh VARCHAR(30),
    @TenChucDanh NVARCHAR(255),
    @CapBac INT = 10,
    @ChucDanhChaID BIGINT = NULL,
    @ThuTu INT = 0,
    @TrangThai TINYINT = 1
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM dbo.DMChucDanh WHERE CongTyID = @CongTyID AND MaChucDanh = @MaChucDanh AND ChucDanhID <> @ChucDanhID AND IsDeleted = 0)
    BEGIN
        RAISERROR(N'Mã chức danh đã tồn tại trong công ty này!', 16, 1);
        RETURN;
    END

    -- Chặn chọn chính nó làm cha (tránh vòng lặp phân cấp)
    IF @ChucDanhChaID IS NOT NULL AND @ChucDanhChaID = @ChucDanhID
    BEGIN
        RAISERROR(N'Chức danh không thể là cha của chính nó!', 16, 1);
        RETURN;
    END

    IF @ChucDanhID = 0
    BEGIN
        INSERT INTO dbo.DMChucDanh (CongTyID, MaChucDanh, TenChucDanh, CapBac, ChucDanhChaID, ThuTu, TrangThai, NgayTao, IsDeleted, Version)
        VALUES (@CongTyID, @MaChucDanh, @TenChucDanh, @CapBac, @ChucDanhChaID, @ThuTu, @TrangThai, SYSUTCDATETIME(), 0, 1);
    END
    ELSE
    BEGIN
        UPDATE dbo.DMChucDanh
        SET MaChucDanh = @MaChucDanh,
            TenChucDanh = @TenChucDanh,
            CapBac = @CapBac,
            ChucDanhChaID = @ChucDanhChaID,
            ThuTu = @ThuTu,

            Version = Version + 1
        WHERE ChucDanhID = @ChucDanhID AND CongTyID = @CongTyID AND IsDeleted = 0;
    END
END;

-- Gán Vai trò cho Tài khoản vừa tạo (hoặc gán lại) — TỰ ĐỘNG chọn đúng cơ chế theo loại Nhóm quyền
CREATE   PROCEDURE dbo.sp_v2_TaiKhoan_GanVaiTro
    @TaiKhoanID INT,
    @NhanVienID INT,
    @NhomQuyenID INT,
    @PhongBanID INT = NULL -- Bắt buộc nếu Vai trò thuộc nhóm PHANCONG (Trưởng phòng/Phó phòng/Tổ trưởng)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @MaNhomQuyen VARCHAR(50) = (SELECT MaNhomQuyen FROM dbo.NhomQuyen WHERE NhomQuyenID = @NhomQuyenID);

    -- Dọn sạch mọi gán quyền cũ của Tài khoản/Nhân viên này trước khi gán mới (tránh chồng chéo)
    UPDATE g SET TrangThai = 0, NgayThuHoi = SYSUTCDATETIME()
    FROM dbo.GanNhomQuyen g
    WHERE (g.ScopeType = 'TAIKHOAN' AND g.TaiKhoanID = @TaiKhoanID)
       OR (g.ScopeType = 'PHANCONG' AND g.PhanCongID IN (SELECT PhanCongID FROM dbo.PhanCongNhanVien WHERE NhanVienID = @NhanVienID));

    UPDATE dbo.PhanCongNhanVien SET TrangThai = 0 WHERE NhanVienID = @NhanVienID;

    IF @MaNhomQuyen IN ('NQ_ADMIN', 'NQ_CN_ADMIN')
    BEGIN
        -- Gán trực tiếp cho Tài khoản
        INSERT INTO dbo.GanNhomQuyen (NhomQuyenID, ScopeType, TaiKhoanID, TrangThai, NgayTao)
        VALUES (@NhomQuyenID, 'TAIKHOAN', @TaiKhoanID, 1, GETDATE());
    END
    ELSE IF @MaNhomQuyen IN ('NQ_TRUONGPHONG', 'NQ_PHOPHONG', 'NQ_TOTRUONG')
    BEGIN
        IF @PhongBanID IS NULL
        BEGIN
            RAISERROR(N'Vui lòng chọn Phòng ban để gán Vai trò này!', 16, 1);
            RETURN;
        END

        DECLARE @LoaiCT INT = (SELECT LoaiPhanCongID FROM dbo.LoaiPhanCong WHERE MaLoai = 'CHINH_THUC');
        DECLARE @ChucVuID INT = (
            SELECT cv.ChucVuID FROM dbo.ChucVu cv
            WHERE cv.MaChucVu = CASE @MaNhomQuyen 
                WHEN 'NQ_TRUONGPHONG' THEN 'TP' 
                WHEN 'NQ_PHOPHONG' THEN 'PP' 
                WHEN 'NQ_TOTRUONG' THEN 'TT' END
        );

        INSERT INTO dbo.PhanCongNhanVien (NhanVienID, LoaiPhanCongID, PhongBanID, ChucVuID, TuNgay, TrangThai, NgayTao)
        VALUES (@NhanVienID, @LoaiCT, @PhongBanID, @ChucVuID, CAST(GETDATE() AS DATE), 1, GETDATE());

        DECLARE @PhanCongID INT = (SELECT TOP 1 PhanCongID FROM dbo.PhanCongNhanVien WHERE NhanVienID = @NhanVienID ORDER BY PhanCongID DESC);

        INSERT INTO dbo.GanNhomQuyen (NhomQuyenID, ScopeType, PhanCongID, TrangThai, NgayTao)
        VALUES (@NhomQuyenID, 'PHANCONG', @PhanCongID, 1, GETDATE());
    END
    -- NQ_NHANVIEN: không cần làm gì thêm — đã được cấp sẵn theo PHONGBAN cho cả phòng ban
END;

GO

CREATE   PROCEDURE dbo.sp_v2_DuAn_Save
    @DuAnID INT = 0,
    @CongTyID INT,
    @PhongBanID INT,
    @LoaiCongTrinhID INT = NULL,
    @MaDuAn VARCHAR(50),
    @TenDuAn NVARCHAR(500),
    @ChuDauTu NVARCHAR(500) = NULL,
    @DiaDiem NVARCHAR(500) = NULL,
    @MoTa NVARCHAR(MAX) = NULL,
    @NgayBatDau DATE = NULL,
    @NgayKetThucDuKien DATE = NULL,
    @TrangThaiDuAn TINYINT = 0,
    @NguoiTaoID INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM dbo.DuAn WHERE CongTyID = @CongTyID AND MaDuAn = @MaDuAn AND DuAnID <> @DuAnID AND IsDeleted = 0)
    BEGIN
        RAISERROR(N'Mã dự án đã tồn tại trong công ty này!', 16, 1);
        RETURN;
    END

    IF @DuAnID = 0
    BEGIN
        INSERT INTO dbo.DuAn (CongTyID, PhongBanID, LoaiCongTrinhID, MaDuAn, TenDuAn, ChuDauTu, DiaDiem, MoTa,
                               NgayBatDau, NgayKetThucDuKien, TienDo, TrangThaiDuAn, QuyTrinhID, NgayTao, IsDeleted, Version, NguoiTaoID)
        VALUES (@CongTyID, @PhongBanID, @LoaiCongTrinhID, @MaDuAn, @TenDuAn, @ChuDauTu, @DiaDiem, @MoTa,
                @NgayBatDau, @NgayKetThucDuKien, 0, @TrangThaiDuAn, NULL, SYSUTCDATETIME(), 0, 1, @NguoiTaoID);

        SELECT SCOPE_IDENTITY() AS NewDuAnID;
    END
    ELSE
    BEGIN
        UPDATE dbo.DuAn
        SET PhongBanID = @PhongBanID,
            LoaiCongTrinhID = @LoaiCongTrinhID,
            MaDuAn = @MaDuAn,
            TenDuAn = @TenDuAn,
            ChuDauTu = @ChuDauTu,
            DiaDiem = @DiaDiem,
            MoTa = @MoTa,
            NgayBatDau = @NgayBatDau,
            NgayKetThucDuKien = @NgayKetThucDuKien,
            TrangThaiDuAn = @TrangThaiDuAn,
            Version = Version + 1
        WHERE DuAnID = @DuAnID AND CongTyID = @CongTyID AND IsDeleted = 0;

        SELECT @DuAnID AS NewDuAnID;
    END
END;

GO

CREATE   PROCEDURE dbo.sp_v2_PhongBan_GetChaOptions
    @ExcludeID INT = 0,
    @ChiNhanhID INT = NULL,
    @ChiNhanhFilterMode BIT = 0
AS
BEGIN
    SET NOCOUNT ON;

    ;WITH ConCha AS (
        SELECT PhongBanID FROM dbo.PhongBan WHERE PhongBanID = @ExcludeID
        UNION ALL
        SELECT pb.PhongBanID FROM dbo.PhongBan pb INNER JOIN ConCha c ON pb.PhongBanChaID = c.PhongBanID
    )
    SELECT PhongBanID, TenPhongBan, CapDo, ChiNhanhID
    FROM dbo.PhongBan
    WHERE IsDeleted = 0
      AND (@ExcludeID = 0 OR PhongBanID NOT IN (SELECT PhongBanID FROM ConCha))
      AND (@ChiNhanhFilterMode = 0 OR (@ChiNhanhID IS NULL AND ChiNhanhID IS NULL) OR ChiNhanhID = @ChiNhanhID)
    ORDER BY CapDo ASC, ThuTu ASC, TenPhongBan ASC;
END;

GO

-- 1.1. Lấy danh sách Phòng ban theo Công ty (có Phân trang, Tìm kiếm, Lọc)
CREATE   PROCEDURE dbo.sp_chinh_DMPhongBan_GetList
    @CongTyID BIGINT,
    @Keyword NVARCHAR(255) = NULL,
    @TrangThai TINYINT = NULL,
    @PageIndex INT = 1,
    @PageSize INT = 20
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
        pb.PhongBanID,
        pb.CongTyID,
        pb.MaPhongBan,
        pb.TenPhongBan,
        pb.PhongBanChaID,
        pbCha.TenPhongBan AS TenPhongBanCha,
        pb.CapDo,
        pb.DuongDan,
        pb.TruongPhongID,
        nv.HoTen AS TenTruongPhong,
        pb.ThuTu,
        pb.TrangThai,
        pb.NgayTao,
        pb.Version,
        COUNT(1) OVER() AS TotalRows
    FROM dbo.DMPhongBan pb
    LEFT JOIN dbo.DMPhongBan pbCha ON pb.PhongBanChaID = pbCha.PhongBanID AND pbCha.IsDeleted = 0
    LEFT JOIN dbo.DMNhanVien nv ON pb.TruongPhongID = nv.NhanVienID AND nv.IsDeleted = 0
    WHERE pb.CongTyID = @CongTyID
      AND pb.IsDeleted = 0
      AND (@TrangThai IS NULL OR pb.TrangThai = @TrangThai)
      AND (
            @Keyword IS NULL 
            OR pb.MaPhongBan LIKE '%' + @Keyword + '%' 
            OR pb.TenPhongBan LIKE '%' + @Keyword + '%'
          )
    ORDER BY pb.CapDo ASC, pb.ThuTu ASC, pb.TenPhongBan ASC
    OFFSET (@PageIndex - 1) * @PageSize ROWS
    FETCH NEXT @PageSize ROWS ONLY;
END;

GO

CREATE   PROCEDURE dbo.sp_v2_PhongBan_Save
    @PhongBanID INT = 0,
    @CongTyID INT,
    @ChiNhanhID INT = NULL,
    @PhongBanChaID INT = NULL,
    @MaPhongBan VARCHAR(50),
    @TenPhongBan NVARCHAR(255),
    @ThuTu INT = 0,
    @TrangThai TINYINT = 1
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM dbo.PhongBan WHERE ChiNhanhID = @ChiNhanhID AND MaPhongBan = @MaPhongBan AND PhongBanID <> @PhongBanID AND IsDeleted = 0
               AND ((@ChiNhanhID IS NULL) OR ChiNhanhID = @ChiNhanhID))
    BEGIN
        RAISERROR(N'Mã phòng ban đã tồn tại trong phạm vi Chi nhánh này!', 16, 1);
        RETURN;
    END

    IF @PhongBanChaID IS NOT NULL AND @PhongBanChaID = @PhongBanID
    BEGIN
        RAISERROR(N'Phòng ban không thể là cha của chính nó!', 16, 1);
        RETURN;
    END

    DECLARE @CapDo INT = 1;
    IF @PhongBanChaID IS NOT NULL
    BEGIN
        SELECT @CapDo = CapDo + 1 FROM dbo.PhongBan WHERE PhongBanID = @PhongBanChaID;
    END

    IF @PhongBanID = 0
    BEGIN
        INSERT INTO dbo.PhongBan (CongTyID, ChiNhanhID, PhongBanChaID, MaPhongBan, TenPhongBan, CapDo, ThuTu, TrangThai, IsDeleted, NgayTao)
        VALUES (@CongTyID, @ChiNhanhID, @PhongBanChaID, @MaPhongBan, @TenPhongBan, @CapDo, @ThuTu, @TrangThai, 0, GETDATE());

        SELECT SCOPE_IDENTITY() AS NewPhongBanID;
    END
    ELSE
    BEGIN
        UPDATE dbo.PhongBan
        SET ChiNhanhID = @ChiNhanhID,
            PhongBanChaID = @PhongBanChaID,
            MaPhongBan = @MaPhongBan,
            TenPhongBan = @TenPhongBan,
            CapDo = @CapDo,
            ThuTu = @ThuTu,
            TrangThai = @TrangThai,
            NgayCapNhat = GETDATE()
        WHERE PhongBanID = @PhongBanID AND IsDeleted = 0;

        SELECT @PhongBanID AS NewPhongBanID;
    END
END;

GO

CREATE   PROCEDURE dbo.sp_v2_BoPhan_GetList
    @Keyword NVARCHAR(255) = NULL,
    @ChiNhanhID INT = NULL,
    @PhongBanID INT = NULL,
    @TrangThai TINYINT = NULL,
    @PageNumber INT = 1,
    @PageSize INT = 12
AS
BEGIN
    SET NOCOUNT ON;

    SELECT COUNT(*) AS TongSoDong
    FROM dbo.PhongBan pb
    WHERE pb.IsDeleted = 0 AND pb.CapDo >= 2
      AND (@Keyword IS NULL OR @Keyword = '' OR pb.MaPhongBan LIKE '%' + @Keyword + '%' OR pb.TenPhongBan LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR pb.PhongBanID = @PhongBanID OR pb.PhongBanChaID = @PhongBanID)
      AND (@TrangThai IS NULL OR pb.TrangThai = @TrangThai);

    SELECT 
        pb.PhongBanID, pb.ChiNhanhID, cn.TenChiNhanh,
        pb.PhongBanChaID, cha.TenPhongBan AS TenPhongBanCha,
        pb.MaPhongBan, pb.TenPhongBan, pb.CapDo, pb.ThuTu, pb.TrangThai
    FROM dbo.PhongBan pb
    LEFT JOIN dbo.ChiNhanh cn ON cn.ChiNhanhID = pb.ChiNhanhID
    LEFT JOIN dbo.PhongBan cha ON cha.PhongBanID = pb.PhongBanChaID
    WHERE pb.IsDeleted = 0 AND pb.CapDo >= 2
      AND (@Keyword IS NULL OR @Keyword = '' OR pb.MaPhongBan LIKE '%' + @Keyword + '%' OR pb.TenPhongBan LIKE N'%' + @Keyword + '%')
      AND (@ChiNhanhID IS NULL OR pb.ChiNhanhID = @ChiNhanhID)
      AND (@PhongBanID IS NULL OR pb.PhongBanID = @PhongBanID OR pb.PhongBanChaID = @PhongBanID)
      AND (@TrangThai IS NULL OR pb.TrangThai = @TrangThai)
    ORDER BY pb.CapDo ASC, pb.ThuTu ASC, pb.TenPhongBan ASC
    OFFSET (@PageNumber - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

GO

CREATE   PROCEDURE dbo.sp_long_DMTaiKhoan_Save
    @TaiKhoanID BIGINT = 0,
    @NhanVienID BIGINT,
    @Username VARCHAR(100),
    @PasswordHash VARCHAR(500) = NULL,  -- NULL khi cập nhật mà không đổi mật khẩu
    @IsLocked BIT = 0,
    @TrangThai TINYINT = 1
AS
BEGIN
    SET NOCOUNT ON;

    IF EXISTS (SELECT 1 FROM dbo.DMTaiKhoan WHERE Username = @Username AND TaiKhoanID <> @TaiKhoanID AND IsDeleted = 0)
    BEGIN
        RAISERROR(N'Tên đăng nhập đã tồn tại trong hệ thống!', 16, 1);
        RETURN;
    END

    IF @TaiKhoanID = 0
    BEGIN
        IF @PasswordHash IS NULL
        BEGIN
            RAISERROR(N'Vui lòng nhập mật khẩu cho tài khoản mới!', 16, 1);
            RETURN;
        END

        INSERT INTO dbo.DMTaiKhoan (NhanVienID, Username, PasswordHash, IsLocked, TrangThai, NgayTao, IsDeleted, Version)
        VALUES (@NhanVienID, @Username, @PasswordHash, @IsLocked, @TrangThai, SYSUTCDATETIME(), 0, 1);

        SELECT SCOPE_IDENTITY() AS NewTaiKhoanID;
    END
    ELSE
    BEGIN
        IF @PasswordHash IS NULL
        BEGIN
            UPDATE dbo.DMTaiKhoan
            SET Username = @Username, IsLocked = @IsLocked, TrangThai = @TrangThai, Version = Version + 1
            WHERE TaiKhoanID = @TaiKhoanID AND IsDeleted = 0;
        END
        ELSE
        BEGIN
            UPDATE dbo.DMTaiKhoan
            SET Username = @Username, PasswordHash = @PasswordHash, IsLocked = @IsLocked, TrangThai = @TrangThai, Version = Version + 1
            WHERE TaiKhoanID = @TaiKhoanID AND IsDeleted = 0;
        END

        SELECT @TaiKhoanID AS NewTaiKhoanID;
    END
END;

CREATE   PROCEDURE dbo.sp_chinh_DMNhanVien_Save
    @NhanVienID BIGINT = 0,               -- 0: Thêm mới, >0: Cập nhật
    @CongTyID BIGINT,
    @PhongBanID BIGINT = NULL,
    @ChucDanhID BIGINT = NULL,
    @QuanLyID BIGINT = NULL,
    @MaNhanVien VARCHAR(30),
    @HoTen NVARCHAR(255),
    @EmailCongTy VARCHAR(255) = NULL,
    @SoDienThoai VARCHAR(20) = NULL,
    @TrangThai TINYINT = 1,
    @NguoiThucHienID BIGINT = NULL        -- Dùng cho Log / Audit
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        -- 1. KIỂM TRA TRÙNG MÃ NHÂN VIÊN TRONG CÙNG CÔNG TY
        IF EXISTS (
            SELECT 1 
            FROM dbo.DMNhanVien 
            WHERE CongTyID = @CongTyID 
              AND MaNhanVien = @MaNhanVien 
              AND IsDeleted = 0 
              AND NhanVienID <> @NhanVienID
        )
        BEGIN
            RAISERROR(N'Mã nhân viên %s đã tồn tại trong công ty này!', 16, 1, @MaNhanVien);
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- 2. THÊM MỚI HOẶC CẬP NHẬT
        IF ISNULL(@NhanVienID, 0) = 0
        BEGIN
            -- INSERT THÔNG TIN NHÂN VIÊN
            INSERT INTO dbo.DMNhanVien (
                CongTyID,
                PhongBanID,
                ChucDanhID,
                QuanLyID,
                MaNhanVien,
                HoTen,
                EmailCongTy,
                SoDienThoai,
                TrangThai,
                IsDeleted,
                Version
            )
            VALUES (
                @CongTyID,
                @PhongBanID,
                @ChucDanhID,
                @QuanLyID,
                @MaNhanVien,
                @HoTen,
                @EmailCongTy,
                @SoDienThoai,
                @TrangThai,
                0, -- IsDeleted
                1  -- Version
            );

            SET @NhanVienID = SCOPE_IDENTITY();

            -- TỰ ĐỘNG THÊM BẢNG NỐI NhanVien_PhongBan (Nếu có chọn phòng ban)
            IF ISNULL(@PhongBanID, 0) > 0
            BEGIN
                INSERT INTO dbo.NhanVien_PhongBan (
                    NhanVienID,
                    PhongBanID,
                    LaChinh,
                    TuNgay
                )
                VALUES (
                    @NhanVienID,
                    @PhongBanID,
                    1, -- LaChinh = true
                    CAST(GETDATE() AS DATE)
                );
            END
        END
        ELSE
        BEGIN
            -- KIỂM TRA SỰ TỒN TẠI
            IF NOT EXISTS (SELECT 1 FROM dbo.DMNhanVien WHERE NhanVienID = @NhanVienID AND IsDeleted = 0)
            BEGIN
                RAISERROR(N'Nhân viên không tồn tại hoặc đã bị xóa!', 16, 1);
                ROLLBACK TRANSACTION;
                RETURN;
            END

            -- LẤY PHÒNG BAN CŨ ĐỂ KIỂM TRA CÓ THAY ĐỔI KHÔNG
            DECLARE @OldPhongBanID BIGINT;
            SELECT @OldPhongBanID = PhongBanID FROM dbo.DMNhanVien WHERE NhanVienID = @NhanVienID;

            -- UPDATE THÔNG TIN NHÂN VIÊN
            UPDATE dbo.DMNhanVien
            SET CongTyID    = @CongTyID,
                PhongBanID  = @PhongBanID,
                ChucDanhID  = @ChucDanhID,
                QuanLyID    = @QuanLyID,
                MaNhanVien  = @MaNhanVien,
                HoTen       = @HoTen,
                EmailCongTy = @EmailCongTy,
                SoDienThoai = @SoDienThoai,
                TrangThai   = @TrangThai,
                Version     = Version + 1
            WHERE NhanVienID = @NhanVienID;

            -- NẾU THAY ĐỔI PHÒNG BAN CHÍNH -> CẬP NHẬT BẢNG NhanVien_PhongBan
            IF ISNULL(@OldPhongBanID, 0) <> ISNULL(@PhongBanID, 0) AND ISNULL(@PhongBanID, 0) > 0
            BEGIN
                -- Bỏ phòng ban chính cũ
                UPDATE dbo.NhanVien_PhongBan
                SET LaChinh = 0
                WHERE NhanVienID = @NhanVienID AND LaChinh = 1;

                -- Thêm phòng ban mới làm phòng ban chính
                INSERT INTO dbo.NhanVien_PhongBan (
                    NhanVienID,
                    PhongBanID,
                    LaChinh,
                    TuNgay
                )
                VALUES (
                    @NhanVienID,
                    @PhongBanID,
                    1,
                    CAST(GETDATE() AS DATE)
                );
            END
        END

        COMMIT TRANSACTION;

        -- TRA VỀ KẾT QUẢ KHỚP 100% VỚI C#
        SELECT 
            @NhanVienID AS NhanVienID,
            1 AS ResponseCode,
            N'Lưu thông tin nhân viên thành công!' AS ResponseMessage;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        RAISERROR(@ErrorMessage, 16, 1);
    END CATCH
END;

GO

CREATE   PROCEDURE dbo.sp_long_DMTaiKhoan_GetNhanVienOptions
    @CongTyID BIGINT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT nv.NhanVienID, nv.HoTen, nv.MaNhanVien
    FROM dbo.DMNhanVien nv
    WHERE nv.CongTyID = @CongTyID
      AND nv.IsDeleted = 0
      AND NOT EXISTS (SELECT 1 FROM dbo.DMTaiKhoan tk WHERE tk.NhanVienID = nv.NhanVienID AND tk.IsDeleted = 0)
    ORDER BY nv.HoTen ASC;
END;

GO


-- 1. SP Lay danh sach / Tim kiem
CREATE   PROCEDURE dbo.sp_chinh_DMCongTy_GetList
    @Keyword NVARCHAR(255) = NULL,
    @TrangThai TINYINT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        CongTyID,
        MaCongTy,
        TenCongTy,
        TenVietTat,
        MaSoThue,
        DiaChi,
        TrangThai,
        MoTa,
        CONVERT(VARCHAR(10), NgayTao, 103) AS NgayTaoText
    FROM dbo.DMCongTy WITH (NOLOCK)
    WHERE IsDeleted = 0
      AND (@Keyword IS NULL OR @Keyword = '' OR MaCongTy LIKE '%' + @Keyword + '%' OR TenCongTy LIKE '%' + @Keyword + '%')
      AND (@TrangThai IS NULL OR TrangThai = @TrangThai)
    ORDER BY CongTyID DESC;
END;

GO

CREATE   PROCEDURE dbo.sp_v2_KiemTraLaAdmin
    @TaiKhoanID INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT CASE WHEN EXISTS (
        SELECT 1 FROM dbo.GanNhomQuyen g
        INNER JOIN dbo.NhomQuyen nq ON nq.NhomQuyenID = g.NhomQuyenID
        WHERE nq.MaNhomQuyen = 'NQ_ADMIN' AND g.ScopeType = 'TAIKHOAN' 
          AND g.TaiKhoanID = @TaiKhoanID AND g.TrangThai = 1
    ) THEN 1 ELSE 0 END AS LaAdmin;
END;

GO


-- ============ LỊCH SỬ QUY TRÌNH (chỉ xem) ============
CREATE   PROCEDURE dbo.sp_long_LichSuQuyTrinh_GetList
    @QuyTrinhID BIGINT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT ls.LichSuID, ls.ObjectType, ls.ObjectID, b.TenBuoc,
           nv.HoTen AS TenNguoiXuLy, ls.KetQua, ls.LyDo,
           CONVERT(VARCHAR(16), ls.NgayXuLy, 120) AS NgayXuLyText
    FROM dbo.LichSuQuyTrinh ls
    INNER JOIN dbo.BuocQuyTrinh b ON b.BuocID = ls.BuocID
    LEFT JOIN dbo.DMNhanVien nv ON nv.NhanVienID = ls.NguoiXuLyID
    WHERE ls.QuyTrinhID = @QuyTrinhID
    ORDER BY ls.NgayXuLy DESC;
END;

GO

CREATE   PROCEDURE dbo.sp_v2_GiaiDoan_GetList
    @Keyword NVARCHAR(255) = NULL
AS
BEGIN

    WHERE IsDeleted = 0

    ORDER BY ThuTuHienThi ASC, TimeLineID ASC;
END;

GO


CREATE   PROCEDURE dbo.sp_long_DMLoaiCongTrinh_GetList
    @CongTyID BIGINT,
    @Keyword NVARCHAR(255) = NULL,
    @TrangThai TINYINT = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SELECT 
        lct.LoaiCongTrinhID, lct.CongTyID, lct.MaLoai, lct.TenLoai,
        lct.LoaiChaID, cha.TenLoai AS TenLoaiCha,
        lct.ThuTu, lct.TrangThai
    FROM dbo.DMLoaiCongTrinh lct
    LEFT JOIN dbo.DMLoaiCongTrinh cha ON cha.LoaiCongTrinhID = lct.LoaiChaID
    WHERE lct.CongTyID = @CongTyID
      AND (@Keyword IS NULL OR @Keyword = '' OR lct.MaLoai LIKE '%' + @Keyword + '%' OR lct.TenLoai LIKE '%' + @Keyword + '%')
      AND (@TrangThai IS NULL OR lct.TrangThai = @TrangThai)
    ORDER BY lct.ThuTu ASC, lct.TenLoai ASC;
END;

GO
